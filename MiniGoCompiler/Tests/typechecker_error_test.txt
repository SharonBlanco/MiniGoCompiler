// =============================================================
// ERROR TEST 5: Arithmetic operator with incompatible types
// =============================================================
func testArithErrors() {
    var a int = 1;
    var s string = "hola";
    var f float64 = 1.5;
    var b bool = true;

    // Addition of different types
    var r1 int = a + f;                 // ERROR: int + float64 incompatible

    // Bitwise operator with non-int
    var r2 float64 = f & f;            // ERROR: bitwise requires int
    var r3 float64 = f | f;            // ERROR: bitwise requires int
    var r4 float64 = f << f;           // ERROR: bitwise requires int
    var r5 float64 = f >> f;           // ERROR: bitwise requires int
    var r6 float64 = f ^ f;            // ERROR: bitwise requires int
    var r7 float64 = f &^ f;          // ERROR: bitwise requires int

    // Unary '-' with string
    var r8 string = -s;                // ERROR: cannot apply - to string

    // Unary '+' with bool
    var r9 bool = +b;                  // ERROR: cannot apply + to bool

    // Unary '^' with float
    var r10 float64 = ^f;             // ERROR: cannot apply ^ to non-int

    // Unary '!' with int
    var r11 int = !a;                  // ERROR: cannot apply ! to non-bool

    // Logical operator with int
    var r12 int = a && a;              // ERROR: && requires bool
    var r13 int = a || a;              // ERROR: || requires bool
};

// =============================================================
// ERROR TEST 7: Compound assignments with incorrect types
// =============================================================
func testAssignErrors() {
    var a int = 1;
    var s string = "hola";
    var b bool = true;

    // += with bool
    b += b;                            // ERROR: cannot apply += to bool

    // += with different types
    a += s;                            // ERROR: incompatible types

    // -= with string
    s -= "x";                          // ERROR: cannot apply -= to string

    // %= with float
    var f float64 = 1.5;
    f %= f;                            // ERROR: cannot apply %= to float

    // &= with float
    f &= f;                           // ERROR: cannot apply &= to float

    // |= with string
    s |= s;                           // ERROR: cannot apply |= to string

    // Assignment = with incompatible types
    a = s;                             // ERROR: incompatible types in assignment
    a, s = s, a;                       // ERROR: incompatible types
};
