func sumar(a int, b int) int {
    return a + b;
};

func restar(a int, b int) int {
    return a - b;
};

func multiplicar(a int, b int) int {
    return a * b;
};

func esPar(n int) bool {
    if n % 2 == 0 {
        return true;
    } else {
        return false;
    };
};

func maximo(a int, b int) int {
    if a > b {
        return a;
    } else {
        return b;
    };
};

func minimo(a int, b int) int {
    if a < b {
        return a;
    } else {
        return b;
    };
};

func clasificar(n int) int {
    switch {
    case n < 0:
        return -1;
    case n == 0:
        return 0;
    default:
        return 1;
    };
};

func letraEn(s string, i int) rune {

func probarShadowing() {
    var x int = 1;
    var texto string = "ab";

    println(`SHADOW 1`);
    println(x);
    println(texto[0], texto[1]);

    {
        var x int = 2;
        var texto string = "xy";

        println(`SHADOW 2`);
        println(x);
        println(texto[0], texto[1]);

        {
            var x int = 3;
            var texto string = "mn";

            println(`SHADOW 3`);
            println(x);
            println(texto[0], texto[1]);
        };

        println(`SHADOW 2 AGAIN`);
        println(x);
        println(texto[0], texto[1]);
    };

    println(`SHADOW 1 AGAIN`);
    println(x);
    println(texto[0], texto[1]);
};
