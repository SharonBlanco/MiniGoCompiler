package main;

// =============================================
// VARIABLES GLOBALES (rúbrica: variables globales)
// =============================================
var gEntero int;
var gFlotante float64;
var gRuna rune;
var gBool bool;

var gA int = 10;
var gB float64 = 3.14;
var gD rune = 'A';
var gE bool = true;

var gX = 42;
var gY = 2.71;
var gW = 'B';
var gV = false;

var gM, gN int = 1, 2;

var (
    gP int = 100;
    gQ float64 = 9.99;
);

// =============================================
// TIPOS Y STRUCTS
// =============================================
type MiEntero int;

type Punto struct {
    x int;
    y int;
};

// =============================================
// ARRAYS (rúbrica: slice y arrays)
// =============================================
var arrGlobal [5]int;

// =============================================
// SLICES (rúbrica: slice y arrays)

var sliceGlobal []int;

// =============================================
// FUNCIONES (rúbrica: métodos globales)
// =============================================

// Procedimiento sin retorno
func imprimir(val int) {
    println(val);
};

// Función con retorno
func suma(a int, b int) int {
    return a + b;
};

// Función con múltiples parámetros
func operar(x int, y int, z float64) int {
    println(x, y, z);
    return x + y;
};

// Función sin parámetros con retorno
func obtenerValor() int {
    return 42;
};

// Función que retorna bool
func esMayor(a int, b int) bool {
    if a > b {
        return true;
    } else {
        return false;
    };
};

// Función con return en switch
func signo(n int) int {
    switch {
    case n > 0:
        return 1;
    case n < 0:
        return -1;
    default:
        return 0;
    };
};

// Función void con return vacío
func saludar() {
    println(`Hola desde MiniGo`);
    return;
};

// =============================================
// MAIN - ejercita todos los puntos de la rúbrica
// =============================================
func main() {

    // -----------------------------------------
    // VARIABLES LOCALES (rúbrica: variables y parámetros locales)
    // -----------------------------------------
    var a int = 5;
    var b int = 10;
    var c int;
    var f float64 = 1.5;
    var g float64 = 2.5;
    var r rune = 'X';
    var flag bool = true;

    // Declaración corta
    x := 99;
    y := 3.14;
    w := 'C';
    v := true;

    // Declaración múltiple
    var m, n int = 100, 200;

    // Bloque parentizado local
    var (
        lp int = 1;
        lq int = 2;
    );

    // -----------------------------------------
    // EXPRESIONES ARITMÉTICAS (rúbrica: expresiones)
    // -----------------------------------------
    c = a + b;
    println(c);
    c = a - b;
    c = a * b;
    c = a / b;
    c = a % b;

    // Float
    var fRes float64 = f + g;
    fRes = f - g;
    fRes = f * g;
    fRes = f / g;
    println(fRes);

    // Bitwise (solo int)
    c = a & b;
    c = a | b;
    c = a ^ b;
    c = a << b;
    c = a >> b;
    c = a &^ b;

    // -----------------------------------------
    // EXPRESIONES RELACIONALES (rúbrica: expresiones)
    // -----------------------------------------
    var cmp bool = a == b;
    cmp = a != b;
    cmp = a < b;
    cmp = a <= b;
    cmp = a > b;
    cmp = a >= b;
    cmp = f == g;
    cmp = f < g;

    // -----------------------------------------
    // EXPRESIONES LÓGICAS (rúbrica: expresiones)
    // -----------------------------------------
    cmp = true && false;
    cmp = true || false;
    cmp = !flag;
    cmp = flag && (a > 0);
    cmp = flag || (b < 10);

    // -----------------------------------------
    // OPERADORES UNARIOS (rúbrica: expresiones)
    // -----------------------------------------
    c = +a;
    c = -a;
    c = ^a;
    var notFlag bool = !flag;

    // Con paréntesis
    c = (a + b) * (a - b);
    println(c);

    // -----------------------------------------
    // ASIGNACIONES COMPUESTAS (rúbrica: expresiones)
    // -----------------------------------------
    c = 10;
    c += a;
    c -= a;
    c *= a;
    c /= a;
    c %= a;
    c &= a;
    c |= a;
    c ^= a;
    c <<= a;
    c >>= a;
    c &^= a;

    fRes = 10.0;
    fRes += f;
    fRes -= f;
    fRes *= f;
    fRes /= f;

    // -----------------------------------------
    // INCREMENTO / DECREMENTO
    // -----------------------------------------
    a++;
    b--;
    f++;
    r--;
    println(a, b, f, r);

    // Asignación múltiple
    a, b = b, a;

    // -----------------------------------------
    // LLAMADAS A FUNCIONES (rúbrica: métodos)
    // -----------------------------------------
    c = suma(a, b);
    println(c);

    var resultado int = suma(10, 20);
    println(resultado);

    imprimir(777);

    resultado = operar(a, b, f);

    resultado = obtenerValor();
    println(resultado);

    flag = esMayor(a, b);
    println(flag);

    var sg int = signo(-5);
    println(sg);

    saludar();

    // -----------------------------------------
    // PRINT Y PRINTLN (rúbrica: métodos predefinidos)
    // -----------------------------------------
    print(a);
    print(a, b, c);
    println(a);
    println(a, f, r, flag);
    println();
    print();
    println(`Esto es un raw string`);

    // -----------------------------------------
    // IF (rúbrica: instrucciones de control)
    // -----------------------------------------

    // If simple
    if a > 0 {
        println(a);
    };

    // If-else
    if a > b {
        println(a);
    } else {
        println(b);
    };

    // If-else if-else
    if a > 10 {
        println(1);
    } else if a > 5 {
        println(2);
    } else {
        println(3);
    };

    // If con init statement
    if val := 10; val > 5 {
        println(val);
    };

    // If con init + else
    if val := 10; val > 15 {
        println(1);
    } else {
        println(0);
    };

    // -----------------------------------------
    // FOR / LOOPS (rúbrica: instrucciones de control)
    // -----------------------------------------

    // For con condición
    var contador int = 0;
    for contador < 5 {
        println(contador);
        contador++;
    };

    // For completo (C-style)
    for i := 0; i < 5; i++ {
        println(i);
    };

    // -----------------------------------------
    // SWITCH (rúbrica: instrucciones de control)
    // -----------------------------------------

    // Switch con expresión
    switch a {
    case 1:
        println(1);
    case 2:
        println(2);
    default:
        println(0);
    };

    // Switch vacío (sin expresión, compara con true)
    switch {
    case a > 0:
        println(a);
    default:
        println(0);
    };

    // Switch con init
    switch val := 5; val {
    case 5:
        println(55);
    default:
        println(0);
    };

    // -----------------------------------------
    // SCOPING (rúbrica: chequeo de alcances)
    // -----------------------------------------
    {
        var a int = 999;
        println(a);
    };
    println(a);

    // -----------------------------------------
    // ARRAYS (rúbrica: slice y arrays)
    // -----------------------------------------
    var arr [5]int;
    arr[0] = 10;
    arr[1] = 20;
    arr[2] = 30;
    arr[3] = 40;
    arr[4] = 50;
    println(arr[0], arr[1], arr[2]);

    // Incremento en array
    arr[0]++;

    // LEN y CAP con array (rúbrica: métodos predefinidos)
    var largo int = len(arr);
    println(largo);
    var capacidad int = cap(arr);
    println(capacidad);

    // -----------------------------------------
    // STRUCTS (rúbrica: tipos de datos)
    // -----------------------------------------
    var p Punto;
    p.x = 10;
    p.y = 20;
    var px int = p.x;
    var py int = p.y;
    println(px, py);

    // -----------------------------------------
    // SLICES con append, len, cap (rúbrica: slice y arrays + métodos predefinidos)
    // -----------------------------------------
    var sl []int;
    sl = append(sl, 100);
    sl = append(sl, 200);
    sl = append(sl, 300);
    var slLen int = len(sl);
    var slCap int = cap(sl);
    println(slLen, slCap);

    // -----------------------------------------
    // RESUMEN de variables usadas
    // -----------------------------------------
    println(m, n, lp, lq, x, w, v);
    println(notFlag, cmp);
    println(gA, gB, gD, gE);
    println(gX, gY, gW, gV);
    println(gM, gN, gP, gQ);
};
