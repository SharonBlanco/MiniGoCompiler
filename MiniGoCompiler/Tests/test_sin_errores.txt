// TEST 14: Función main - ejercita todo lo demás
// =============================================================
func main() {

    // --- Declaraciones locales ---
    var a int = 1;
    var b int = 2;
    var c int;
    var f float64 = 1.5;
    var g float64 = 2.5;
    var s string = "test";
    var r rune = 'X';
    var flag bool = true;

    // --- Expresiones aritméticas ---
    c = a + b;
    c = a - b;
    c = a * b;
    c = a / b;
    c = a % b;

    // --- Expresiones aritméticas con float ---
    var fResult float64 = f + g;
    fResult = f - g;
    fResult = f * g;
    fResult = f / g;

    // --- Expresiones bitwise (solo int) ---
    c = a & b;
    c = a | b;
    c = a ^ b;
    c = a << b;
    c = a >> b;
    c = a &^ b;

    // --- Expresiones relacionales ---
    var cmp bool = a == b;
    cmp = a != b;
    cmp = a < b;
    cmp = a <= b;
    cmp = a > b;
    cmp = a >= b;

    // --- Comparación de floats ---
    cmp = f == g;
    cmp = f < g;

    // --- Comparación de strings ---
    cmp = s == "otro";
    cmp = s != "otro";

    // --- Expresiones lógicas ---
    cmp = true && false;
    cmp = true || false;
    cmp = !flag;
    cmp = flag && (a > 0);
    cmp = flag || (b < 10);

    // --- Operadores unarios ---
    c = +a;
    c = -a;
    c = ^a;
    var notFlag bool = !flag;

    // --- Expresión con paréntesis ---
    c = (a + b) * (a - b);

    // --- Asignaciones compuestas ---
    c += a;
    c -= a;
    c *= a;
    c /= a;
    c %= a;
    c &= a;
    c |= a;
    c ^= a;
    c <<= a;
    c >>= a;
    c &^= a;

    // --- += con string (concatenación) ---
    s += "extra";

    // --- += con float ---
    fResult += f;
    fResult -= f;
    fResult *= f;
    fResult /= f;

    // --- Incremento y decremento ---
    a++;
    b--;
    f++;
    r--;

    // --- Declaración corta := ---
    x := 99;
    y := 3.14;
    z := "corta";
    w := 'C';
    v := true;

    // --- Asignación múltiple ---
    a, b = b, a;

    // --- Llamada a función ---
    c = suma(a, b);
    var resultado int = suma(10, 20);

    // --- Llamada a procedimiento ---
    procedimiento();

    // --- Llamada a función con múltiples tipos ---
    resultado = mezcla(a, f, s);

    // --- Llamada a función que retorna bool ---
    flag = esPositivo(a);

    // --- Print y Println ---
    print(a);
    print(a, b, c);
    println(a);
    println(a, f, s, r, flag);
    println();
    print();

    // --- If simple ---
    if a > 0 {
        println(a);
    };

    // --- If-else ---
    if a > 0 {
        println("positivo");
    } else {
        println("no positivo");
    };

    // --- If-else if-else ---
    if a > 0 {
        println("positivo");
    } else if a == 0 {
        println("cero");
    } else {
        println("negativo");
    };

    // --- If con init statement ---
    if x := 10; x > 5 {
        println(x);
    };

    // --- If con init + else ---
    if x := 10; x > 5 {
        println("si");
    } else {
        println("no");
    };

    // --- For infinito ---
    for {
        break;
    };

    // --- For con condición ---
    for a < 100 {
        a++;
        if a > 50 {
            break;
        };
    };

    // --- For completo (C-style) ---
    for i := 0; i < 10; i++ {
        println(i);
    };

    // --- For sin condición ---
    for i := 0;; i++ {
        if i > 5 {
            break;
        };
    };

    // --- For con continue ---
    for i := 0; i < 10; i++ {
        if i == 5 {
            continue;
        };
        println(i);
    };

    // --- Switch con expresión ---
    switch a {
    case 1:
        println("uno");
    case 2:
        println("dos");
    default:
        println("otro");
    };

    // --- Switch vacío (sin expresión) ---
    switch {
    case a > 0:
        println("positivo");
    default:
        println("no positivo");
    };

    // --- Switch con init ---
    switch x := 5; x {
    case 5:
        println("cinco");
    default:
        println("otro");
    };

    // --- Bloque anidado (scoping) ---
    {
        var a int = 999;
        println(a);
    };
    println(a);

    // --- Arrays: acceso e indexación ---
    var arr [3]int;
    arr[0] = 10;
    arr[1] = 20;
    arr[2] = 30;
    var elem int = arr[0];
    println(elem);

    // --- Len con array ---
    var largo int = len(arr);
    println(largo);

    // --- Len con string ---
    largo = len(s);

    // --- Cap con array ---
    var capacidad int = cap(arr);

    // --- Structs ---
    var p Punto;
    p.x = 10;
    p.y = 20;
    var px int = p.x;
    println(px, p.y);

    // --- Slices con append ---
    var sl []int;
    sl = append(sl, 1);
    sl = append(sl, 2);
    var slElem int = sl[0];

    // --- Len y Cap con slice ---
    largo = len(sl);
    capacidad = cap(sl);

    // --- String indexing retorna rune ---
    var ch rune = s[0];
    println(ch);

    // --- Type declaration local ---
    type LocalType int;

    // --- Variable declaration en bloque ---
    var (
        lp int = 1;
        lq int = 2;
    );
    println(lp, lq);

    // --- Retorno implícito de main (void) ---
};

// TEST 15: Función con return garantizado vía if-else
// =============================================================
func absoluto(n int) int {
    if n >= 0 {
        return n;
    } else {
        return -n;
    };
};

// =============================================================
// TEST 16: Función con return garantizado vía switch
// =============================================================
func signo(n int) int {
    switch {
    case n > 0:
        return 1;
    case n < 0:
        return -1;
    default:
        return 0;
    };
};
